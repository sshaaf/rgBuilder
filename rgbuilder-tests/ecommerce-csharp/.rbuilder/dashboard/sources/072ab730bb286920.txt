using Ecommerce.Models;
using Microsoft.EntityFrameworkCore;

namespace Ecommerce.Data;

public static class DataSeeder
{
    public static async Task SeedAsync(EcommerceDbContext db)
    {
        if (await db.Categories.AnyAsync())
        {
            return;
        }

        var electronics = new Category
        {
            Name = "Electronics",
            Description = "Gadgets and devices"
        };
        db.Categories.Add(electronics);
        await db.SaveChangesAsync();

        var laptop = new Product
        {
            Name = "Demo Laptop",
            Description = "Lightweight laptop for everyday use",
            Price = 999.99m,
            CategoryId = electronics.Id,
            Inventory = new Inventory { Quantity = 25 }
        };
        db.Products.Add(laptop);

        var headphones = new Product
        {
            Name = "Demo Headphones",
            Description = "Wireless noise-cancelling headphones",
            Price = 149.99m,
            CategoryId = electronics.Id,
            Inventory = new Inventory { Quantity = 100 }
        };
        db.Products.Add(headphones);

        await db.SaveChangesAsync();
    }
}
